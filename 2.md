/* STREAMING_CHUNK:Registering DOM element selectors and state variables... */
// DOM 요소 가져오기
const orderForm = document.getElementById('orderForm');
const userNameInput = document.getElementById('userName');
const userPhoneInput = document.getElementById('userPhone');
const drinkSelect = document.getElementById('drinkSelect');
const sizeRadios = document.querySelectorAll('input[name="size"]');
const optionCheckboxes = document.querySelectorAll('input[name="option"]');
const quantityInput = document.getElementById('quantity');
const requestsInput = document.getElementById('requests');

const totalPriceElement = document.getElementById('totalPrice');
const confirmMessage = document.getElementById('confirmMessage');

/* STREAMING_CHUNK:Implementing price calculation function... */
/**
 * 실시간 예상 금액을 계산하여 화면에 표시하는 함수
 */
function calculatePrice() {
    let price = 0;

    // 1. 음료 단가 계산
    const selectedOption = drinkSelect.options[drinkSelect.selectedIndex];
    const drinkPrice = parseInt(selectedOption.getAttribute('data-price') || 0, 10);
    price += drinkPrice;

    // 음료가 선택된 경우에만 사이즈, 옵션, 수량 금액 계산
    if (drinkPrice > 0) {
        // 2. 사이즈 추가 금액 계산
        sizeRadios.forEach(radio => {
            if (radio.checked) {
                price += parseInt(radio.getAttribute('data-price') || 0, 10);
            }
        });

        // 3. 추가 옵션 금액 계산
        optionCheckboxes.forEach(checkbox => {
            if (checkbox.checked) {
                price += parseInt(checkbox.getAttribute('data-price') || 0, 10);
            }
        });

        // 4. 수량 곱하기
        const quantity = parseInt(quantityInput.value, 10) || 1;
        price *= quantity;
    }

    // 천 단위 콤마 표기 적용 후 화면 업데이트
    totalPriceElement.textContent = `${price.toLocaleString('ko-KR')}원`;
    return price;
}

/* STREAMING_CHUNK:Attaching real-time event listeners for price updating... */
// 실시간 금액 업데이트 이벤트 리스너 등록
drinkSelect.addEventListener('change', calculatePrice);

sizeRadios.forEach(radio => {
    radio.addEventListener('change', calculatePrice);
});

optionCheckboxes.forEach(checkbox => {
    checkbox.addEventListener('change', calculatePrice);
});

quantityInput.addEventListener('input', () => {
    // 수량 유효 범위 처리 (1 ~ 10)
    if (quantityInput.value < 1) quantityInput.value = 1;
    if (quantityInput.value > 10) quantityInput.value = 10;
    calculatePrice();
});

/* STREAMING_CHUNK:Implementing Excel/CSV download helper function... */
/**
 * 주문 내역을 엑셀(CSV) 파일로 다운로드하는 함수
 */
function exportToExcel(orderData) {
    // CSV 헤더 및 데이터 생성
    const headers = ["주문시간", "이름", "전화번호", "음료명", "사이즈", "추가옵션", "수량", "요청사항", "총금액"];
    const row = [
        `"${orderData.time}"`,
        `"${orderData.name}"`,
        `"${orderData.phone}"`,
        `"${orderData.drink}"`,
        `"${orderData.size}"`,
        `"${orderData.options}"`,
        `"${orderData.quantity}"`,
        `"${orderData.requests}"`,
        `"${orderData.totalPrice.toLocaleString('ko-KR')}원"`
    ];

    // 한글 깨짐 방지를 위한 BOM (\uFEFF) 선언
    const csvContent = "\uFEFF" + headers.join(",") + "\n" + row.join(",");

    // Blob 객체 생성 및 파일 다운로드 링크 트리가링
    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    
    // 파일명 지정 (예: 바이브카페_주문서_홍길동.csv)
    link.setAttribute('href', url);
    link.setAttribute('download', `바이브카페_주문서_${orderData.name}.csv`);
    document.body.appendChild(link);
    
    link.click();
    document.body.removeChild(link);
}

/* STREAMING_CHUNK:Handling form submission and reset events... */
// 폼 제출(주문하기 버튼 클릭) 처리
orderForm.addEventListener('submit', (e) => {
    e.preventDefault();

    // 1. 이름 유효성 검사
    const nameVal = userNameInput.value.trim();
    if (!nameVal) {
        alert("이름을 입력해주세요");
        userNameInput.focus();
        return;
    }

    // 2. 음료 선택 유효성 검사
    const drinkVal = drinkSelect.value;
    if (!drinkVal) {
        alert("음료를 선택해주세요");
        drinkSelect.focus();
        return;
    }

    // 사이즈 및 옵션 선택 값 정리
    let sizeVal = "M";
    sizeRadios.forEach(radio => {
        if (radio.checked) sizeVal = radio.value;
    });

    const selectedOptions = [];
    optionCheckboxes.forEach(checkbox => {
        if (checkbox.checked) selectedOptions.push(checkbox.value);
    });

    const optionsText = selectedOptions.length > 0 ? ` (${selectedOptions.join(', ')})` : '';
    const quantityVal = quantityInput.value;
    const finalPrice = calculatePrice();

    // 3. 주문 확인 메시지 표시
    // 예시: "홍길동님, 카페라떼 M사이즈 (샷 추가) 1잔, 총 5,000원 주문이 접수되었습니다!"
    confirmMessage.textContent = `${nameVal}님, ${drinkVal} ${sizeVal}사이즈${optionsText} ${quantityVal}잔, 총 ${finalPrice.toLocaleString('ko-KR')}원 주문이 접수되었습니다!`;
    confirmMessage.classList.remove('hidden');

    // 4. 엑셀(CSV) 파일로 주문 정보 저장
    const now = new Date();
    const timeFormatted = `${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,'0')}-${String(now.getDate()).padStart(2,'0')} ${String(now.getHours()).padStart(2,'0')}:${String(now.getMinutes()).padStart(2,'0')}`;

    const orderData = {
        time: timeFormatted,
        name: nameVal,
        phone: userPhoneInput.value.trim() || '미입력',
        drink: drinkVal,
        size: sizeVal,
        options: selectedOptions.join(', ') || '없음',
        quantity: quantityVal,
        requests: requestsInput.value.trim() || '없음',
        totalPrice: finalPrice
    };

    exportToExcel(orderData);
});

// 다시 작성 버튼 클릭 처리 (폼 리셋 후 초기화)
orderForm.addEventListener('reset', () => {
    setTimeout(() => {
        calculatePrice(); // 금액 0원으로 리셋
        confirmMessage.classList.add('hidden'); // 주문 메시지 감추기
        confirmMessage.textContent = '';
    }, 10);
});

// 초기화 시 금액 계산 한번 실행
calculatePrice();
